52class JsonFormatter(logging.Formatter): 

53 def format(self, record: logging.LogRecord) -> str: 

54 payload: dict[str, Any] = { 

55 "timestamp": datetime.now(UTC).isoformat(), 

56 "level": record.levelname.lower(), 

57 "service": "enhanceai-core-backend", 

58 "message": record.getMessage(), 

59 } 

60 

61 # Keep every extra field the caller attached (output_bytes, 

62 # generate_seconds, realtime_factor, cloned, faces_detected, path, …), 

63 # not just a fixed allowlist. Values must be JSON-serialisable; anything 

64 # exotic is stringified so a stray object never drops the whole line. 

65 for key, value in record.__dict__.items(): 

66 if key in _STANDARD_RECORD_KEYS or key.startswith("_") or value is None: 

67 continue 

68 try: 

69 json.dumps(value) 

70 payload[key] = value 

71 except (TypeError, ValueError): 

72 payload[key] = str(value) 

73 

74 if record.exc_info: 

75 payload["exception"] = self.formatException(record.exc_info) 

76 

77 return json.dumps(payload)
